cmake_minimum_required(VERSION 3.22)
project(SwaraXTCutoffParserContract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(APPLE)
 enable_language(OBJC OBJCXX)
endif()
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${JUCE_ROOT}" JUCE EXCLUDE_FROM_ALL)
add_library(CutoffTestJuce STATIC)
target_link_libraries(CutoffTestJuce PRIVATE juce::juce_audio_processors)
target_compile_definitions(CutoffTestJuce PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-include")
file(WRITE "${CMAKE_BINARY_DIR}/test-include/JuceHeader.h" "#pragma once\n#include <juce_audio_processors/juce_audio_processors.h>\n")
foreach(variant original portable)
 add_executable(Cutoff-${variant})
 if(variant STREQUAL "original")
  set(layout "${ORIGINAL_LAYOUT}")
 else()
  set(layout "${SW2_ROOT}/Source/Plugin/SwaraXtParameterLayout.cpp")
  target_compile_definitions(Cutoff-${variant} PRIVATE SWARAXT_FORCE_PORTABLE_REAL_PARSER=1)
 endif()
 target_sources(Cutoff-${variant} PRIVATE CutoffParameterTests.cpp "${layout}")
 target_include_directories(Cutoff-${variant} PRIVATE "${CMAKE_BINARY_DIR}/test-include" "${SW2_ROOT}/Source" "${SW2_ROOT}/Source/Engine/ShruthiPort/include_first" "${SW2_ROOT}/Source/Engine/ShruthiPort/adapted" "${SW2_ROOT}/third_party/shruthi-1" "${JUCE_ROOT}/modules")
 target_compile_definitions(Cutoff-${variant} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(Cutoff-${variant} PRIVATE CutoffTestJuce)
 target_compile_options(Cutoff-${variant} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
